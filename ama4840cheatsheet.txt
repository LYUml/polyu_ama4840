\section*{Topic 1: Prob \& Bayes}

\subsection*{1. Basic Probability}
Total Prob: $P(A) = \sum_i P(A|B_i)P(B_i)$ \\
Cond Prob: $P(A|B) = \frac{P(A \cap B)}{P(B)}$ \\
Indep: $P(A \cap B) = P(A)P(B) \Leftrightarrow P(A|B) = P(A)$

\subsection*{2. Distributions \& Math Core}
\textbf{Binomial (Discrete):} $X \sim B(n, p)$ \\
$P(X=r) = C_n^r p^r (1-p)^{n-r}$ \\
\textbf{Poisson (Discrete):} $X \sim \text{Poi}(\lambda)$ \\
$P(X=x) = \frac{e^{-\lambda} \lambda^x}{x!} \quad \text{Total: } \sum\nolimits_{x=0}^{\infty} \frac{\lambda^x}{x!} = e^\lambda$ \\
\textbf{Poisson CDF (Sum to $k$):} $P(X \le k) = \sum\nolimits_{x=0}^{k} \frac{e^{-\lambda} \lambda^x}{x!}$ \\
\textbf{Poisson-Gamma Integral Link (Shortcut):} \\
\textit{(Used to bypass integrating $\lambda$ in Bayesian posterior)} \\
$\int_0^c \frac{\lambda^n e^{-\lambda}}{n!} d\lambda = 1 - \sum\nolimits_{x=0}^n \frac{c^x e^{-c}}{x!} = 1 - P(Y \le n \mid \lambda=c)$ \\
\textbf{General Form (for $\lambda$ from $0$ to $c$):} \\
$\int_0^c \lambda^n e^{-a\lambda} d\lambda = \frac{n!}{a^{n+1}} \left[ 1 - \sum\nolimits_{x=0}^n \frac{(ac)^x e^{-ac}}{x!} \right]$

\textbf{Exponential (Cont.):} $X \sim \text{Exp}(\lambda)$ \\
$f(x) = \lambda e^{-\lambda x} \quad (x \ge 0)$; $E[X] = \frac{1}{\lambda}, Var(X) = \frac{1}{\lambda^2}$ \\
\textbf{Gamma / Integration Shortcut:} \\
$\int_0^{\infty} \lambda^n e^{-a\lambda} d\lambda = \frac{n!}{a^{n+1}}$ \\
\textit{($\lambda$: rate, $n$: total count, $a$: total time)} \\
Int by Parts: $\int u dv = uv - \int v du$ \\
$\int x^n e^{-ax} dx = -e^{-ax} \sum\nolimits_{k=0}^{n} \frac{n! x^{n-k}}{k! a^{k+1}}$ \textit{ (Compact Form)}

\textbf{Exp CDF (Area):} $\int_0^c \lambda e^{-\lambda x} dx = 1 - e^{-\lambda c}$ \\
\textbf{Exp Tail (Survive):} $\int_c^\infty \lambda e^{-\lambda x} dx = e^{-\lambda c}$ \\
\textbf{Linear Shift ($u$-sub Bypass):} \\
$\int e^{ax+b} dx = \frac{1}{a} e^{ax+b}, \quad \int (ax+b)^n dx = \frac{(ax+b)^{n+1}}{a(n+1)}$

\subsection*{3. Discrete Bayesian (All PMFs)}
\textit{Note: All $P(\cdot)$ here are PMFs (Discrete points).} \\
Bayes: $P(S_i | I) = \frac{P(I | S_i)P(S_i)}{\sum_j P(I | S_j)P(S_j)}$ \\
Tabular Method: \\
1. Prior \textbf{PMF}: $P(S_i)$ \\
2. Likelihood \textbf{PMF}: $P(I|S_i)$ \\
3. Joint \textbf{PMF}: $P(S_i \cap I) = P(I|S_i)P(S_i)$ \\
4. Marginal \textbf{PMF}: $P(I) = \sum \text{Joint}$ \\
5. Posterior \textbf{PMF}: $P(S_i | I) = \text{Joint} / \text{Marginal}$

\subsection*{4. Binomial Sampling Bayes (All PMFs)}
\textit{Note: Though $f$ is used, binomial is discrete $\to$ PMF.} \\
Likelihood \textbf{PMF}: $f(r | n, p_i) = C_n^r p_i^r (1-p_i)^{n-r}$ \\
Posterior \textbf{PMF}:
$$P(p_i | r, n) = \frac{p_i^r (1-p_i)^{n-r} P(p_i)}{\sum_j p_j^r (1-p_j)^{n-r} P(p_j)}$$

\subsection*{5. Continuous Bayesian (All PDFs)}
\textit{Note: All $f(\cdot)$ here are PDFs. Use Integration ($\int$) to find probabilities.} \\
Prior \textbf{PDF}: $f_{\Lambda}(\lambda)$ \\
Likelihood \textbf{PDF}: $f(x|\lambda)$ \\
Marginal(unconditional) \textbf{PDF}: $f_X(x) = \int f(x|\lambda) f_{\Lambda}(\lambda) d\lambda$ \\
Posterior \textbf{PDF}:
$$f_{\Lambda}(\lambda | x) = \frac{f(x|\lambda) f_{\Lambda}(\lambda)}{\int f(x|\lambda) f_{\Lambda}(\lambda) d\lambda}$$
Expected Value: $E[\lambda | x] = \int \lambda \cdot f_{\Lambda}(\lambda | x) d\lambda$


\section*{Topic 2: Decision Analysis \& Sampling}

\subsection*{Decision Analysis: Universal Bayesian Updating}
\textbf{The 4-Step Table Method} (Works for ANY evidence/distribution): \\
\textbf{1. Prior ($P(H_i)$):} Initial probability of each state. \textit{(Must sum to 1)} \\
\textbf{2. Likelihood ($P(E|H_i)$):} Prob. of seeing the new Evidence IF in state $H_i$. \textit{(Calculated via given rules/distributions. Do NOT sum to 1)} \\
\textbf{3. Joint:} $\mathbf{Prior \times Likelihood}$. The combined weight. \\
\textbf{4. Posterior:} $\mathbf{Joint / \sum Joint}$. The updated probability. \textit{(Normalized to sum to 1)} \\
\textit{*Crucial: Always use the Posterior probabilities to calculate the NEW EMV for your final decision.*}

\subsection*{1. Decision Under Uncertainty (No Probs)}
\textbf{Maximax:} $\max (\max \text{Payoff})$ (Optimistic) \\
\textbf{Maximin:} $\max (\min \text{Payoff})$ (Pessimistic) \\
\textbf{Hurwicz (Realism):} $\alpha(\text{Max in Row}) + (1-\alpha)(\text{Min in Row})$ \\
\textbf{Minimax Regret:} Regret = (Best in Col) - Payoff. Choose act with $\min (\max \text{Regret})$. \\
\textbf{Laplace:} Average all payoffs in the row. Choose Max.

\subsection*{2. Prior Analysis (Before Sample)}
\textbf{EMV$_{\text{prior}}$:} Expected Monetary Value $= \sum [ P_{\text{prior}}(s_i) \cdot \text{Payoff}(s_i) ]$. Choose Max. \\
\textbf{EOL$_{\text{prior}}$:} Expected Opp Loss $= \sum [ P_{\text{prior}}(s_i) \cdot \text{Regret}(s_i) ]$. Choose Min. \\
\textit{Optimal Act $a^0$ has Max EMV$_{\text{prior}}$ (which equals Min EOL$_{\text{prior}}$).}

\subsection*{3. Value of Perfect Info (EVPI)}
Upper bound for any research cost. \\
\textbf{EVWPI} (Expected Value With Perfect Info) $= \sum [ P_{\text{prior}}(s_i) \cdot (\text{Best Payoff in } s_i) ]$. \\
\textbf{EVPI} $= \text{EVWPI} - \max(\text{EMV}_{\text{prior}})$. \\
\textit{Shortcut: EVPI is exactly equal to $\min(\text{EOL}_{\text{prior}})$ of optimal act $a^0$!}

\subsection*{4. Pipeline: EVSI (Buying Imperfect Info $z$)}
When you buy imperfect info (Sample/Survey $z$). \\
\textbf{Step 1 (Prior):} Find optimal act $a^0$ and its $\text{EMV}_{\text{prior}}$ without sample. \\
\textbf{Step 2 (Sample Prob $P(z)$):} Use Bayes to find total probability of getting survey result $P(z)$. \\
\textbf{Step 3 (Posterior EMV):} For a specific result $z$, update state probs to $\mathbf{P_{\text{post}}(s_i|z)}$. Calculate new EMVs. Find optimal act $a_z^0$ and its $\mathbf{\text{EMV}_{\text{post}}(z)}$. \\
\textbf{Step 4 (EMV With Sample):} Multiply each optimal posterior EMV by the probability of getting that survey result: \\
$\text{EMVS} = \sum [ P(z) \cdot \text{EMV}_{\text{post}}(z) ]$. \\
\textbf{Step 5 (EVSI \& ENGS):} \\
\textbf{EVSI} $= \text{EMVS} - \max(\text{EMV}_{\text{prior}})$. \\
\textbf{ENGS} (Expected Net Gain) $= \text{EVSI} - \text{Cost of Sample}$. \\
\textit{Rule: If ENGS $> 0$, buy the sample information!}

\subsection*{5. Marginal Analysis (Discrete)}
Let $P$: Prob(Success), $k$: Cost of under-ordering, $h$: Cost of over-ordering. \\
Optimal quantity $Q^*$ satisfies: $P(D < Q^*) = \frac{k}{k+h}$

\section*{Topic 3: Utility Theory (Single)}

\subsection*{1. Risk Attitudes}
Let $U(x)$ be the utility function, $x$ be the payoff.\\
\textbf{Risk Averse:} $U''(x) < 0$ (Concave curve) \\
Prefers sure amount over lottery. $CE < E[X]$ \\
\textbf{Risk Neutral:} $U''(x) = 0$ (Linear line) \\
Indifferent. $CE = E[X]$ \\
\textbf{Risk Seeking:} $U''(x) > 0$ (Convex curve) \\
Prefers lottery over sure amount. $CE > E[X]$

\subsection*{2. Core Metrics}
For a lottery $L$ with payoffs $x_i$ and probabilities $p_i$: \\
\textbf{Expected Value (EMV):} $E[X] = \sum p_i x_i$ \\
\textbf{Expected Utility:} $E[U(X)] = \sum p_i U(x_i)$ \\
\textbf{Certainty Equivalent (CE):} \\
The guaranteed return having the same utility as $L$. \\
$U(CE) = E[U(X)] \implies CE = U^{-1}(E[U(X)])$ \\
\textbf{Risk Premium ($\pi$):} \\
The amount "paid" to avoid risk. \\
$\pi = E[X] - CE$ \\
\textit{Risk Averse $\Rightarrow \pi > 0$; Risk Seeking $\Rightarrow \pi < 0$}

\subsection*{3. Lottery Pricing (Buying vs Selling)}
Let $W$ = Current Wealth (Asset Position), $L$ = Lottery. \\
\textbf{Buying Price ($B$):} \\
Max amount willing to pay to acquire the right to play $L$. \\
$U(W) = E[U(W - B + L)]$ \\
\textit{(Utility of staying with current wealth = Expected utility of paying B and playing L)} \\
\textbf{Selling Price ($S$):} \\
Min amount willing to accept to give up the right to play $L$. \\
$U(W + S) = E[U(W + L)]$ \\
\textit{(Utility of taking cash S and walking away = Expected utility of playing L with current wealth)}

\subsection*{4. Common Utility Functions \& Price Properties}
\textbf{1. Exponential Utility:} $U(x) = 1 - e^{-x/R}$ ($R > 0$) \\
\textit{Property: \textbf{CARA} (Constant Absolute Risk Aversion).} \\
\textbf{Comment:} Exhibits the Delta Property. Buying Price = Selling Price ($B=S$). Strictly independent of initial wealth $W$.

\textbf{2. Logarithmic Utility:} $U(x) = \ln(x)$ ($x > 0$) \\
\textit{Property: \textbf{DARA} (Decreasing Absolute Risk Aversion).} \\
\textbf{Comment:} Exhibits the Wealth Effect. As wealth increases, risk aversion decreases. Therefore, Selling Price $>$ Buying Price ($S>B$). Depends on $W$.

\textbf{3. Quadratic Utility:} $U(x) = ax - bx^2$ ($b > 0, x \le a/2b$) \\
\textit{Property: \textbf{IARA} (Increasing Absolute Risk Aversion).} \\
\textbf{Comment:} As wealth increases, risk aversion increases (economically counter-intuitive). Therefore, Buying Price $>$ Selling Price ($B>S$). Depends on $W$.

\subsection*{5. Indifference Lotteries}
To find an unknown parameter (e.g., $R$ or $\alpha$): \\
If a person is "indifferent" between $L_1$ and $L_2$: \\
Set $E[U(L_1)] = E[U(L_2)]$ and solve for the unknown.


\section*{Topic 4: Multi-Attribute Utility}

\subsection*{1. Independence Conditions}
Let $u(x, y)$ be utility of attribute $X$ and $Y$. \\
\textbf{Additive Independence (AI):} \\
$u(x, y) = k_x u_x(x) + k_y u_y(y)$ where $k_x + k_y = 1$. \\
\textit{Condition: Preferences over lotteries on X,Y depend only on marginal distributions.} \\
\textbf{Utility Independence (UI):} \\
$X$ is UI of $Y$ if preferences over lotteries on $X$ do not depend on fixed $y$. \\
\textbf{Mutual Utility Independence (MUI):} \\
$X$ is UI of $Y$ AND $Y$ is UI of $X$.

\subsection*{2. Multiplicative Form (MUI)}
If $X, Y$ are MUI, total utility is: \\
$u(x, y) = k_x u_x(x) + k_y u_y(y) + k u_x(x) u_y(y)$ \\
where $k = 1 - k_x - k_y$. \\
\textbf{Scaling:} $u(x^*, y^*) = 1$, $u(x^0, y^0) = 0$. \\
$u_i(x_i^*) = 1$, $u_i(x_i^0) = 0$. \\
\textbf{Finding $k_i$:} $k_x = u(x^*, y^0)$, $k_y = u(x^0, y^*)$.

\subsection*{3. Evaluation Steps}
Step 1: Define $u_x(x)$ and $u_y(y)$ based on Topic 3 (Risk Attitude). \\
Step 2: Solve for $k_x, k_y$ using boundary conditions or indifference points. \\
Step 3: Determine $k = 1 - k_x - k_y$. \\
Step 4: Calculate $u(x,y)$ for all choices and rank by $E[u(x,y)]$.

\section*{Topic 5: Analytic Hierarchy (AHP)}

\subsection*{1. Minimize CI \& Unknowns (KFC Yiu's Trick)}
\textbf{1. The 3x3 Transitive Rule (Primary):} \\
To minimize CI ($CI \to 0$), force perfect consistency: \\
$\mathbf{a_{ik} = a_{ij} \times a_{jk}}$ \textit{(e.g., $a_{13} = a_{12} \times a_{23}$, solve for $x, y$)} \\
\textit{Proof: By AM-GM inequality, $3 \times 3$ $\lambda_{\max}$ is minimized when sum of reciprocal cycles $k+1/k \ge 2$ (where $k = \frac{a_{12} \times a_{23}}{a_{13}}$) reaches minimum at $k=1$.} \\
\textbf{2. The 1-9 Scale Boundary (Edge Case):} \\
Saaty scale is strictly 1 to 9. If Transitive Rule yields $a_{ij} > 9$ (e.g., $4 \times 3 = 12$), you MUST cap it: $\mathbf{a_{ij} = 9}$ (closest valid num). \\
\textbf{3. The 4x4 Multi-Path Conflict:} \\
If reaching node $D$ from $A$ yields conflicting path results (e.g., via $B \to 6$, via $C \to 8$), use \textbf{Geometric Mean}: \\
$a_{14}^* = \sqrt{\text{Path 1} \times \text{Path 2}} = \sqrt{6 \times 8} \approx 6.93$.

\subsection*{2. Standard 6-Step Calculation (e.g., 3x3)}
Given Matrix $A$. \\
\textbf{Step 1: Col Sum ($C_j$)} \\
$C_1 = a_{11}+a_{21}+a_{31}$, $C_2 = \dots$, $C_3 = \dots$ \\
\textbf{Step 2: Normalize ($a'_{ij}$)} \\
Divide each element by its Col Sum: $a'_{ij} = a_{ij} / C_j$ \\
\textbf{Step 3: Weights Vector ($w_i$)} \\
Row average of Normalized Matrix: \\
$w_1 = (a'_{11} + a'_{12} + a'_{13}) / 3$ (Same for $w_2, w_3$) \\
\textbf{Step 4: Weighted Sum Vector ($v_i$)} \\
Multiply original $A$ by $w$: \\
$v_1 = a_{11}w_1 + a_{12}w_2 + a_{13}w_3$ (Same for $v_2, v_3$) \\
\textbf{Step 5: Max Eigenvalue ($\lambda_{\max}$)} \\
$\lambda_{\max} = \frac{1}{3} (v_1/w_1 + v_2/w_2 + v_3/w_3)$ \\
\textbf{Step 6: Test CI and CR} \\
$CI = \frac{\lambda_{\max} - n}{n - 1}$ ($3*3 \rightarrow n=3$) \\
$CR = CI / RI$ (Accept if $< 0.10$. $RI_{n=3}=0.58, RI_{n=4}=0.90$)

\subsection*{3. Reverse Engineering (Weights $\to$ Matrix)}
If $w_1, w_2, w_3$ are given and Matrix $A$ is unknown: \\
Assume perfectly consistent: $a_{ij} = w_i / w_j$. \\
\textit{(e.g., $a_{12} = w_1 / w_2$, $a_{31} = w_3 / w_1$)}


\section*{Topic 6: Game Theory (Pipeline)}

\textbf{Two-Person Zero-Sum Game}
\begin{itemize}
    \item Payoff matrix $A = [a_{ij}]$: amount Player I wins from Player II.
    \item $\sum\nolimits$ payoffs = $0$ for all outcomes.
\end{itemize}

\textbf{Gain Floor \& Loss Ceiling (exist for ALL games)} \\
Gain Floor (maximin): $v_I' = \max\nolimits_{i}\left\{\min\nolimits_{j} a_{ij}\right\}$ \\
Loss Ceiling (minimax): $v_{II}' = \min\nolimits_{j}\left\{\max\nolimits_{i} a_{ij}\right\}$

\textbf{Expected Gain Floor \& Expected Loss Ceiling (Mixed Strategies)} \\
Expected Gain Floor: $v_I = \max\nolimits_{x}\left\{\min\nolimits_{j} x A_{\cdot j}\right\}$ \\
Expected Loss Ceiling: $v_{II} = \min\nolimits_{y}\left\{\max\nolimits_{i} A_{i\cdot} y^T\right\}$

\textbf{Minimax Inequality}: 
$v_I' \le v_I = v_{II} \le v_{II}'$ \\
If $v_I' = v_{II}' = v$, then a \textbf{saddle point} exists.

\textbf{Saddle Point}:
An entry $a_{ij}$ that is the \textbf{minimum in its row} and \textbf{maximum in its column}.

\textbf{Minimax Theorem}:
Every finite two-person zero-sum game has optimal mixed strategies, and 
$\max\nolimits_{x}\min\nolimits_{y} xAy^T = \min\nolimits_{y}\max\nolimits_{x} xAy^T = v$
where $v$ is the \textbf{value of the game}.

\textbf{Dominance}:
\begin{itemize}
    \item Row $k$ dominates Row $l$ if $a_{kj} \ge a_{lj}$ for all $j$.
    \item Column $k$ dominates Column $l$ if $a_{ik} \le a_{il}$ for all $i$.
\end{itemize}

\subsection*{Step 1: Find Saddle Point (Pure)}
1. \textbf{Row (Maximin):} Find min of each row $\to$ Pick the MAX. \\
2. \textbf{Col (Minimax):} Find max of each col $\to$ Pick the MIN. \\
\textit{If Maximin = Minimax = $v$, stop! Optimal is pure (1s and 0s).}

\subsection*{Step 2: Dominance (Reduce Matrix)}
\textbf{Row wants MAX:} If Row A $\ge$ Row B $\implies$ \textbf{DELETE SMALLER} Row B. \\
\textbf{Col wants MIN:} If Col A $\le$ Col B $\implies$ \textbf{DELETE LARGER} Col B. \\
\textit{Repeat until 2x2. Remember which rows/cols you deleted!}

\subsection*{Step 3: Solve 2x2 (Matrix Method \& Shortcut)}
Matrix $A = [a, b; c, d]$. Let $J = [1, 1]$, $A^* = [d, -b; -c, a]$, $|A| = ad - bc$. \\
\textbf{Formulas:} $x = \frac{JA^*}{JA^*J^T}, \quad y = \frac{J(A^*)^T}{JA^*J^T}, \quad v = \frac{|A|}{JA^*J^T}$ \\
\textit{(Shortcut: Denom $\Delta = a+d-b-c$. \\
Row $x \implies$ sum cols/$\Delta$. Col $y \implies$ sum rows/$\Delta$.)} \\
\textbf{eg.:} $A = \left[ \begin{smallmatrix} 4 & 3 \\ 3 & 5 \end{smallmatrix} \right] \implies \text{Swap \& Negate} \implies A^* = \left[ \begin{smallmatrix} 5 & -3 \\ -3 & 4 \end{smallmatrix} \right]$. $\Delta = 5-3-3+4 = \mathbf{3}$. \\
\textbf{Row $x$ (Sum Cols of $A^*$ $\downarrow$):} Col 1 $= 5-3=2$, Col 2 $= -3+4=1 \implies \mathbf{x = [2/3, 1/3]}$. \\
\textbf{Col $y$ (Sum Rows of $A^*$ $\rightarrow$):} Row 1 $= 5-3=2$, Row 2 $= -3+4=1 \implies \mathbf{y = [2/3, 1/3]}$.

\subsection*{Step 4: Restore Original Vector (Add 0s)}
Map $x, y$ back to original size to get $x^*, y^*$. \\
Assign $0$ to any strategy that was deleted in Step 2. \\
\textit{Example: If Row 3 was deleted, $x^* = (x_1, x_2, 0)$.}

\subsection*{Statistical Games, Loss and Decision Functions}
\textbf{1. Mathematical Definitions}
\begin{itemize}
    \item \textbf{Decision Function}: A rule $d(x)$ mapping observation $x$ to an action $a$.
    \item \textbf{Loss Function}: $L(d(x), \theta)$ is the cost incurred when taking action $d(x)$ given true state $\theta$.
    \item \textbf{Risk Function}: $R(d, \theta) = E_{X|\theta} [L(d(X), \theta)] = \sum\nolimits_{x} L(d(x), \theta) \cdot P(X=x \mid \theta)$. \\
    (Risk is the \textbf{average loss} of rule $d$ across all $x$ for a fixed $\theta$).
    \item \textbf{Minimax Criterion}: $\min\nolimits_{d} \max\nolimits_{\theta} R(d, \theta)$. Choose $d$ to minimize the maximum possible risk.
    \item \textbf{Equalizer Rule}: A rule $d^*$ where $R(d^*, \theta) = \text{C}$ (constant) for all $\theta$.
\end{itemize}

\textbf{2. Risk Matrix Example (Coin Decision)}
\textit{Setup: Nature $\theta_1$(2-head), $\theta_2$(Bal.); Decision $d_i$ (Logic rules).}
\begin{center}
\setlength{\tabcolsep}{2pt}
\tiny
\begin{tabular}{|c|c|c|c|c|} \hline
$R(d, \theta)$ & $d_1$ & $d_2$ & $d_3$ & $d_4$ \\ \hline
$\theta_1$ & 0 & 0 & 1 & 1 \\ \hline
$\theta_2$ & 0.5 & 1.0 & 0 & 0.5 \\ \hline
\textbf{Max Risk} & \textbf{0.5} & 1.0 & 1.0 & 1.0 \\ \hline
\end{tabular}
\end{center}
\textbf{Minimax Choice}: Rule $d_1$ (it has the smallest maximum risk: 0.5).

\textbf{3. Calculation Shortcut (Equalizer Principle)}
To find optimal constants ($a, b$) for an estimator $d(x) = \frac{x+a}{n+b}$:
\begin{itemize}
    \item \textbf{Step 1}: Write out the Risk Function $R(d, \theta)$ in terms of $\theta$.
    \item \textbf{Step 2}: \textbf{Force Constant Risk}: Set all coefficients of $\theta$ and $\theta^2$ in the $R(d, \theta)$ expression to \textbf{zero}.
    \item \textbf{Step 3}: Solve the resulting equations for $a$ and $b$.
\end{itemize}
\textit{Note: This makes the risk "spyproof" and independent of Nature's state.}

Let $y=e^{500/R}$. $y^{-2}+y=2 \implies y^3-2y^2+1=0$. Root $y = \frac{1+\sqrt{5}}{2} \approx 1.618$.
$500/R = \ln(1.618) = 0.481 \implies R = 1039$. Nearest 100 is $\mathbf{R=1000}$.
\textit{(ii)} $u(x) = 1-e^{-x/1000}$. CARA utility implies Buy Price = Sell Price.
Expected Utility of $L_1$ with W=1000: $E[u] = 0.6 u(2000) + 0.4 u(500) = 1 - (0.6e^{-2} + 0.4e^{-0.5}) \approx 0.6762$. (Note: assume $L_1$ shifted to 0.6/0.4 based on text).
CE: $1-e^{-CE/1000} = 0.6762 \implies e^{-CE/1000} = 0.3238 \implies CE = 1127.6$.
Buy/Sell Price = $1127.6 - 1000 = \mathbf{\$127.6}$. They are equal due to CARA.
